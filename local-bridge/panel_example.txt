        <span id="projectDetails">Đang kiểm tra…</span>
      </div>
      <p class="hint">Mở cùng thư mục này trong Codex desktop để xem file. Diff và hoàn tác từng đợt dùng được cả khi thư mục chưa có Git.</p>

<footer class="foot">

<script>
(() => {
  "use strict";
  const $ = id => document.getElementById(id), pending = new Map(), UI_VERSION = "2.1.0";
  const statuses = {
    prepared: "Đang chờ áp dụng",
    applied: "Đã áp dụng",
    undone: "Đã hoàn tác",
    applying: "Đang ghi",
    undoing: "Đang hoàn tác",
    recovered: "Đã phục hồi sau lỗi",
    recovery_conflict: "Cần xử lý xung đột",
    undo_failed: "Hoàn tác lỗi, đã phục hồi",
    expired: "Đã hết thời hạn lưu bản sao",
    running: "Đang chạy",
    succeeded: "Thành công",
    failed: "Thất bại",
    stopped: "Đã dừng",
    timed_out: "Hết thời gian",
    runtime_lost: "Mất kết nối bộ chạy"
  };

  let connected = false, disposed = false, visible = true, hostOrigin = null,
      project = null, change = null, runId = null, cursor = 0, timer = null,
      polling = false, busy = false, nextDiff = null, pendingStart = null;

  const notify = (method, params) => parent.postMessage({ jsonrpc: "2.0", method, params }, hostOrigin || "*");

  function rpc(method, params) {
    return new Promise((resolve, reject) => {
      const id = crypto.randomUUID(),
            timeout = setTimeout(() => {
              pending.delete(id);
              reject(new Error("Kết nối hết thời gian. Thao tác chưa được tự chạy lại; bấm Làm mới để kiểm tra."));
            }, 60000);
      pending.set(id, { resolve, reject, timeout });
      parent.postMessage({ jsonrpc: "2.0", id, method, params }, hostOrigin || "*");
    });
  }

  function unpack(result) {
    if (result?.isError) throw new Error(result.content?.filter(x => x.type === "text").map(x => x.text).join("\n") || "Tool báo lỗi.");
    if (result?.structuredContent) return result.structuredContent;
    const text = result?.content?.find(x => x.type === "text")?.text || "{}";
    try { return JSON.parse(text); } catch { return {}; }
  }

  async function call(name, args = {}) {
    return unpack(await rpc("tools/call", { name, arguments: args }));
  }

  function error(e) {
    const message = e?.message || String(e);
    $("error").textContent = message;
    $("error").hidden = false;
    $("connection").dataset.state = "error";
    $("connectionText").textContent = "Cần kiểm tra kết nối";
  }

  function clearError() {
    $("error").textContent = "";
    $("error").hidden = true;
  }

  function options(select, items, key, label, empty) {
    const value = select.value;
    select.replaceChildren();
    if (empty) {
      const o = document.createElement("option");
      o.value = "";
      o.textContent = empty;
      select.append(o);
    }
    for (const item of items) {
      const o = document.createElement("option");
      o.value = item[key];
      o.textContent = label(item);
      select.append(o);
    }
    if ([...select.options].some(o => o.value === value)) select.value = value;
  }

  function statusPill(text, kind = "") {
    const span = document.createElement("span");
    span.className = "status-pill" + (kind ? " " + kind : "");
    span.textContent = text;
    return span;
  }

  function buttons() {
    const active = !!project?.active_run_id;
    const runtime = project?.runtime || {};
    $("apply").disabled = busy || active || change?.status !== "prepared";
    $("undo").disabled = busy || active || change?.status !== "applied";
    $("start").disabled = busy || active || !project?.runtime?.commands_enabled || !$("taskSelect").value;
    $("stop").disabled = busy || !active;
    $("modeNormal").disabled = busy || active || runtime.status !== "connected" || runtime.mode === "normal";
    $("modeTurbo").disabled = busy || active || runtime.status !== "connected" || runtime.mode === "turbo";
  }

  function renderProject(p) {
    project = p;
    const runtime = p.runtime || {}, mode = runtime.mode === "turbo" ? "turbo" : "normal",
          ready = !!runtime.commands_enabled, active = !!p.active_run_id;
    $("projectPath").textContent = p.workspace_root || "Không xác định";
    $("connection").dataset.state = runtime.status === "connected" ? "connected" : "connecting";
    $("connectionText").textContent = runtime.status === "connected" ? "Đã kết nối" : "Bộ chạy chưa kết nối";
    $("projectDetails").textContent = `${runtime.version || "Chưa có phiên bản Codex"} · ${p.git_repository ? "Có Git" : "Thư mục chưa có Git"}`;
    $("modeBadge").textContent = mode === "turbo" ? "Turbo" : "Normal";
    $("modeBadge").className = `status-pill${mode === "turbo" ? " danger" : ""}`;
    $("modeNormal").setAttribute("aria-pressed", String(mode === "normal"));
    $("modeTurbo").setAttribute("aria-pressed", String(mode === "turbo"));
    $("modeDescription").textContent = runtime.mode_description || (mode === "turbo" ? "Turbo đang bật." : "Normal đang bật.");
    $("runtimeMetric").textContent = mode === "turbo" ? "Turbo" : ready ? "Sẵn sàng" : "Giới hạn";
    $("runtimeHint").textContent = mode === "turbo" ? "Toàn quyền Bash / command" : (ready ? "Có thể chạy task đã duyệt" : (runtime.command_error || "Doctor chưa đạt"));
    $("taskMetric").textContent = active ? "Đang chạy" : (ready ? "Sẵn sàng" : "Tắt");
    $("taskHint").textContent = active ? "Một task đang giữ khóa ghi" : (ready ? "Tối đa một task" : "Chức năng chạy đang khóa");
    $("taskNote").textContent = mode === "turbo" ? "Turbo cho phép task và Bash dùng quyền toàn máy; chỉ chạy lệnh mày đã yêu cầu." : ready ? "Tối đa một task; file được khóa ghi trong lúc chạy." : "Bộ chạy chưa đạt kiểm tra bảo vệ dữ liệu và mạng. Chức năng chạy task đang tạm khóa; xem chi tiết bằng doctor.";
    if (p.active_run_id && p.active_run_id !== runId) {
      runId = p.active_run_id;
      cursor = 0;
      $("logs").textContent = "";
      $("logSummary").textContent = "Đang nhận log…";
    }
    buttons();
    schedule();
  }

  function renderChanges(items) {
    $("changeCount").textContent = items.length;
    $("changeHint").textContent = items.length ? "Chọn một đợt để xem diff" : "Chưa có đợt nào";
    options($("changeSelect"), items, "id", x => `${x.title} · ${statuses[x.status] || x.status}`, "Chọn đợt sửa");
  }

  function renderTasks(data) {
    const tasks = data?.tasks || [];
    options($("taskSelect"), tasks, "task_id", x => x.task_id);
    if (!tasks.length) $("taskSelect").replaceChildren(Object.assign(document.createElement("option"), { value: "", textContent: "Chưa cấu hình task" }));
    buttons();
  }

  function renderDiffLines(diffText, append = false) {
    const diffEl = $("diff");
    if (!diffText && !append) {
      diffEl.textContent = "Không có diff.";
      return;
    }
    const fragment = document.createDocumentFragment();
    const lines = (diffText || "").split(/\r?\n/);
    for (let i = 0; i < lines.length; i++) {
      const line = lines[i];
      if (i === lines.length - 1 && line === "" && !append) continue;
      const lineEl = document.createElement("div");
      lineEl.className = "diff-line";
      if (line.startsWith("+++") || line.startsWith("---")) {
        lineEl.classList.add("diff-file");
      } else if (line.startsWith("@@")) {
        lineEl.classList.add("diff-hunk");
      } else if (line.startsWith("+")) {
        lineEl.classList.add("diff-add");
      } else if (line.startsWith("-")) {
        lineEl.classList.add("diff-del");
      }
      // textContent guarantees complete XSS immunity
      lineEl.textContent = line || " ";
      fragment.appendChild(lineEl);
    }
    if (!append) {
      diffEl.replaceChildren(fragment);
    } else {
      diffEl.appendChild(fragment);
    }
  }

  function clearChange() {
    change = null;
    nextDiff = null;
    $("fileSelect").replaceChildren(Object.assign(document.createElement("option"), { value: "", textContent: "Tất cả file" }));
    $("changeStatus").replaceChildren(statusPill("Chưa chọn"), document.createTextNode(" Chuẩn bị thay đổi trong hội thoại để xem diff ở đây."));
    $("diff").textContent = "Chưa chọn đợt sửa.";
    $("moreDiff").classList.add("hidden");
    buttons();
  }

  async function refresh() {
    const [p, c, t] = await Promise.all([call("project_info"), call("list_changes"), call("list_tasks")]);
    renderProject(p);
    renderChanges(c.changes || []);
    renderTasks(t);
    if ($("changeSelect").value) await loadChange();
    else clearChange();
    if (runId) await pollRun();
    buttons();
  }

  async function loadChange(append = false) {
    const id = $("changeSelect").value;
    if (!id) {
      clearChange();
      return;
    }
    const selected = $("fileSelect").value,
          result = await call("get_change", { change_id: id, path: selected || null, offset: append ? (nextDiff || 0) : 0 });
    change = result;
    nextDiff = result.next_offset;
    const kind = result.status === "recovery_conflict" || result.status === "undo_failed" ? "danger"
               : result.status === "prepared" || result.status === "applying" ? "warn" : "";
    $("changeStatus").replaceChildren(
      statusPill(statuses[result.status] || result.status, kind),
      document.createTextNode(` · ${result.change_id}${result.error ? " · " + result.error : ""}`)
    );
    options($("fileSelect"), result.files || [], "path", x => x.path, "Tất cả file");
    renderDiffLines(result.diff, append);
    $("moreDiff").classList.toggle("hidden", nextDiff == null);
    buttons();
  }

  function snapshot(data) {
    if (data.project) renderProject(data.project);
    if (Array.isArray(data.changes)) renderChanges(data.changes);
    if (data.task_state) renderTasks(data.task_state);
  }

  async function mutation(name) {
    if (!change) return;
    await call(name, { change_id: change.change_id, request_id: `ui-${name}-${change.change_id}` });
    await refresh();
  }

  async function start() {
    const task = $("taskSelect").value;
    if (!pendingStart) pendingStart = { task_id: task, request_id: crypto.randomUUID(), timeout_seconds: 120 };
    const run = await call("start_task", pendingStart);
    pendingStart = null;
    runId = run.run_id;
    cursor = 0;
    $("logs").textContent = "";
    $("logSummary").textContent = "Đang nhận log…";
    await refresh();
  }

  async function switchMode(mode, confirm = false) {
    await call("set_runtime_mode", { mode, confirm });
    $("turboConfirm").hidden = true;
    await refresh();
  }

  async function pollRun() {
    if (!runId) return;
    const r = await call("get_task_run", { run_id: runId, cursor }),
          text = (r.events || []).map(e => e.text).join("");
    if (text) $("logs").textContent = ($("logs").textContent + text).slice(-262144);
    cursor = r.next_cursor;
    const running = r.status === "running";
    $("taskStatus").replaceChildren(
      statusPill(`${statuses[r.status] || r.status}${r.stopping ? " · đang dừng…" : ""}`,
                 running ? "warn" : r.status === "failed" || r.status === "runtime_lost" ? "danger" : "")
    );
    $("taskProgressWrap").hidden = !running;
    $("taskProgress").setAttribute("aria-valuetext", running ? "Đang xử lý" : statuses[r.status] || r.status);
    $("logSummary").textContent = `${r.logs_available ? "Log đang lưu" : "Log đã hết hạn"}${r.truncated ? " · đã cắt" : ""} · ${r.next_cursor} dòng sự kiện`;
    if (r.error) $("taskNote").textContent = r.error;
    if (!running && project) {
      project.active_run_id = null;
      $("taskMetric").textContent = statuses[r.status] || r.status;
      buttons();
    }
  }

  function schedule() {
    clearTimeout(timer);
    if (!disposed && connected && visible && !document.hidden && project?.active_run_id) {
      timer = setTimeout(async () => {
        if (polling) { schedule(); return; }
        polling = true;
        try {
          await pollRun();
          renderProject(await call("project_info"));
        } catch (e) {
          error(e);
        } finally {
          polling = false;
          schedule();
        }
      }, 2000);
    }
  }

  async function action(fn) {
    busy = true;
    buttons();
    clearError();
    try { await fn(); }
    catch (e) { error(e); }
    finally {
      busy = false;
      buttons();
      schedule();
    }
  }

  $("refresh").onclick = () => action(refresh);
  $("changeSelect").onchange = () => action(async () => { $("fileSelect").value = ""; await loadChange(); });
  $("fileSelect").onchange = () => action(() => loadChange());
  $("moreDiff").onclick = () => action(() => loadChange(true));
  $("apply").onclick = () => action(() => mutation("apply_changes"));
  $("undo").onclick = () => action(() => mutation("undo_changes"));
  $("start").onclick = () => action(start);
  $("modeNormal").onclick = () => action(() => switchMode("normal"));
  $("modeTurbo").onclick = () => {
    if (!busy && !project?.active_run_id) $("turboConfirm").hidden = false;
  };
  $("modeCancel").onclick = () => { $("turboConfirm").hidden = true; };
  $("modeConfirmButton").onclick = () => action(() => switchMode("turbo", true));
  $("stop").onclick = () => action(async () => {
    await call("stop_task_run", { run_id: runId, request_id: `ui-stop-${runId}` });
    await pollRun();
  });

  function applyHostContext(context) {
    if (context?.theme) {
      document.documentElement.style.colorScheme = context.theme;
      document.documentElement.setAttribute("data-theme", context.theme);
    }
    const vars = context?.styles?.variables || {};
    for (const [key, value] of Object.entries(vars)) {
      if (/^--[a-z0-9-]{1,80}$/i.test(key) && typeof value === "string" && value.length <= 200) {
        document.documentElement.style.setProperty(key, value);
      }
    }
  }

  window.addEventListener("message", event => {
    if (event.source !== parent || !event.data || event.data.jsonrpc !== "2.0" || (hostOrigin && event.origin !== hostOrigin)) return;
    const m = event.data;
    if (pending.has(m.id)) {
      const p = pending.get(m.id);
      pending.delete(m.id);
      clearTimeout(p.timeout);
      if (!hostOrigin && event.origin !== "null") hostOrigin = event.origin;
      m.error ? p.reject(new Error(m.error.message)) : p.resolve(m.result);
      return;
    }
    if (m.method === "ui/notifications/tool-result") {
      try { snapshot(unpack(m.params || {})); } catch (e) { error(e); }
    } else if (m.method === "ui/notifications/tool-input") {
      const input = m.params?.arguments || m.params?.input || {};
      if (input.change_id) {
        $("changeSelect").value = input.change_id;
        action(() => loadChange());
      }
    } else if (m.method === "ui/notifications/tool-cancelled") {
      error(new Error(m.params?.reason || "Thao tác bị hủy."));
    } else if (m.method === "ui/resource-teardown") {
      disposed = true;
      clearTimeout(timer);
      if (m.id != null) parent.postMessage({ jsonrpc: "2.0", id: m.id, result: {} }, hostOrigin || "*");
    } else if (m.method === "ui/notifications/request-teardown") {
      disposed = true;
      clearTimeout(timer);
    } else if (m.method === "ui/notifications/host-context-changed") {
      applyHostContext(m.params || {});
    }
  });

  document.addEventListener("visibilitychange", schedule);
  new IntersectionObserver(entries => { visible = entries[0].isIntersecting; schedule(); }).observe(document.body);
  new ResizeObserver(() => {
    if (connected) notify("ui/notifications/size-changed", { height: document.documentElement.scrollHeight });
  }).observe(document.body);

  rpc("ui/initialize", {
    protocolVersion: "2026-01-26",
    appInfo: { name: "local-bridge-panel", version: UI_VERSION },
    appCapabilities: { tools: { listChanged: false }, availableDisplayModes: ["inline", "fullscreen"] }
  }).then(result => {
    connected = true;
    $("connection").dataset.state = "connected";
    notify("ui/notifications/initialized", {});
    applyHostContext(result?.hostContext || {});
    return action(refresh);
  }).catch(error);
})();
</script>
